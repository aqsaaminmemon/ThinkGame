cmake_minimum_required(VERSION 3.28)
project(ThinkGame VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(THINKGAME_BUILD_APP "Build the graphical game" ON)
option(THINKGAME_BUILD_UI_TESTS "Build render/event tests (requires a graphical display)" OFF)
include(CTest)
add_library(thinkgame_core src/puzzle.cpp src/progress.cpp src/game.cpp src/assets.cpp)
target_include_directories(thinkgame_core PUBLIC include)
if(MSVC)
  target_compile_options(thinkgame_core PRIVATE /W4)
else()
  target_compile_options(thinkgame_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(thinkgame_tests tests/core_tests.cpp)
target_link_libraries(thinkgame_tests PRIVATE thinkgame_core)
add_test(NAME rules_progress_saves COMMAND thinkgame_tests)
add_executable(thinkgame_catalogue src/catalogue.cpp)
target_link_libraries(thinkgame_catalogue PRIVATE thinkgame_core)
if(THINKGAME_BUILD_APP)
  find_package(SFML 3 QUIET COMPONENTS Graphics Window System Audio)
  if(NOT SFML_FOUND)
    include(FetchContent)
    set(SFML_BUILD_AUDIO ON CACHE BOOL "" FORCE)
    set(SFML_BUILD_NETWORK OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_TEST_SUITE OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    # Official 3.0.2 tag verified at 0fa201c969e48ecc253581c5841ce73f44d42f49.
    FetchContent_Declare(SFML GIT_REPOSITORY https://github.com/SFML/SFML.git GIT_TAG 3.0.2 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(SFML)
  endif()
  add_executable(thinkgame src/main.cpp src/ui.cpp src/audio.cpp)
  target_link_libraries(thinkgame PRIVATE thinkgame_core SFML::Graphics SFML::Audio)
  target_compile_definitions(thinkgame PRIVATE THINKGAME_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
  add_custom_command(TARGET thinkgame POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:thinkgame>/assets")
  install(TARGETS thinkgame RUNTIME DESTINATION bin)
  install(DIRECTORY assets/ DESTINATION share/thinkgame/assets)
  if(THINKGAME_BUILD_UI_TESTS)
    add_executable(thinkgame_ui_tests tests/ui_tests.cpp src/ui.cpp src/audio.cpp)
    target_link_libraries(thinkgame_ui_tests PRIVATE thinkgame_core SFML::Graphics SFML::Audio)
    target_compile_definitions(thinkgame_ui_tests PRIVATE THINKGAME_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
    add_test(NAME actual_graphics_and_events COMMAND thinkgame_ui_tests)
  endif()
endif()
